% Basiert auf KOMA-Scripts scrlttr2

\documentclass[
	% Schriftgröße
	fontsize=12pt,
	%
	% zwischen Absätzen eine leere Zeile einfügen, statt lediglich Einrückung
	parskip=full,
	%
	% Papierformat auf DIN-A4
	paper=A4,
	%
	% Kontaktdaten stehen im Informationsblock statt im Briefkopf
	fromphone=false,
	fromfax=false,
	fromemail=false,
	fromurl=false,
	%
	% Faltmarkierungen verbergen
	%foldmarks=false,
	%
	% Die neuste Version von scrlettr2 verwenden
	version=last,
]{scrlttr2}

% Briefform B; für Form A hier DIN5008A einsetzen
\LoadLetterOption{DIN5008B}

% Name und Anschrift weiterhin oben rechts, weitere Angaben im Informationsblock
\KOMAoptions{fromalign=right,firsthead=true,refline=nodate}
\setplength{firstheadwidth}{\textwidth}
\removereffields

% Zeichenkodierung des Dokuments ist in UTF-8
\usepackage[utf8]{inputenc}

% Eurosymbol-Unterstützung
\usepackage{eurosym}
% Das Unicode-Zeichen € als \euro interpretieren.
% So kann man direkt € tippen anstatt jedes Mal \euro auszuschreiben.
\DeclareUnicodeCharacter{20AC}{\euro}

% Sprache des Dokuments auf Deutsch
\usepackage[ngerman]{babel}

% Includen von PDFs nach dem Brief, siehe \includepdf unten
\usepackage{pdfpages}

% klickbare Links und E-Mail-Adressen. Paket url kann keine klickbaren,
% deswegen hyperref. Option hidelinks versteckt farbigen Rahmen.
\usepackage[hidelinks]{hyperref}

% Absendername unter Schlussformel entfernen. Dieser wird bereits aus dem Briefkopf ersichtlich.
% Hier wird die signature-Variable einfach auf einen leeren Wert gesetzt und wäre sonst \usekomavar{fromname}.
\setkomavar{signature}{}

% Für Schlussformel (und nicht vorhandenen Namen darunter) Linksbündigkeit erzwingen
%\renewcommand*{\raggedsignature}{\raggedright}

% Informationsblock rechts neben dem Anschriftfeld; leere Angaben entfallen
\setkomavar{location}{%
	\raggedright\usekomafont{fromaddress}%
	\setlength{\parskip}{0pt}%
	\Ifkomavarempty{yourref}{}{\usekomavar*{yourref}: \usekomavar{yourref}\par}%
	\Ifkomavarempty{yourmail}{}{\usekomavar*{yourmail}: \usekomavar{yourmail}\par}%
	\Ifkomavarempty{myref}{}{\usekomavar*{myref}: \usekomavar{myref}\par}%
	\Ifkomavarempty{customer}{}{\usekomavar*{customer}: \usekomavar{customer}\par}%
	\Ifkomavarempty{invoice}{}{\usekomavar*{invoice}: \usekomavar{invoice}\par}%
	\Ifkomavarempty{fromphone}{}{\usekomavar*{fromphone}\usekomavar{fromphone}\par}%
	\Ifkomavarempty{fromfax}{}{\usekomavar*{fromfax}\usekomavar{fromfax}\par}%
	\Ifkomavarempty{fromemail}{}{\usekomavar*{fromemail}\usekomavar{fromemail}\par}%
	\Ifkomavarempty{fromurl}{}{\usekomavar*{fromurl}\usekomavar{fromurl}\par}%
	\Ifkomavarempty{date}{}{\usekomavar*{date}: \usekomavar{date}\par}%
}
